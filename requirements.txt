# Neight — runtime dependencies
#
#   python -m venv .venv
#   .venv\Scripts\activate          (Windows)
#   source .venv/bin/activate       (macOS / Linux)
#   pip install -r requirements.txt
#
# Versions are pinned so a release can be reproduced.  They record what Neight
# is actually built and tested against, not a lower bound: `PySide6>=6.0.0`
# meant every machine linked a different Qt, and a Qt minor release can change
# text layout, cursor geometry and Tamil shaping without a line of code changing
# here.  Bump deliberately, then re-run the cross-platform checks.
#
# Development, build and design tools live in requirements-dev.txt, so a runtime
# install no longer drags in PyInstaller and Pillow.

PySide6==6.11.0
shiboken6==6.11.0
